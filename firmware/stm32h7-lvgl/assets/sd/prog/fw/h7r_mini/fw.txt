# STM32H7R mini — 내부 FSBL + 외부 QSPI 펌웨어
#
# 이 보드는 내부 플래시가 64KB 뿐이라 부트로더(FSBL)만 들어가고, 실제 펌웨어는
# 외부 QSPI(W25Q128JV, 0x90000000)에 있다. 그래서 알고리즘이 둘 필요하다.
#
#   image 가 어느 쪽으로 갈지는 주소로 정해진다. 적을 문법이 없다.
#     fsbl.elf  p_paddr 0x08000000  -> 내부 (DB 의 algo)
#     app.elf   p_paddr 0x90000000  -> loader

name   = STM32H7R mini (FSBL + QSPI)

# 외부 로더는 여러 번 적을 수 있다. 담당 주소가 겹치면 거부한다.
loader = /prog/loaders/ext/W25Q128JV_STM32H7R-MINI.stldr

image  = fsbl.elf
image  = app.elf

# 소거·굽기 병렬도. ST FlashLoader 만 쓰고 외부 로더는 무시한다.
psize  = 2
